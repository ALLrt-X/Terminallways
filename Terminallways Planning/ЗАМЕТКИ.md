private static readonly HttpClient Http = CreateHttpClient();

private static HttpClient CreateHttpClient()
{
    var handler = new HttpClientHandler
    {
        ServerCertificateCustomValidationCallback = (req, cert, chain, errors) =>
        {
            if (errors == SslPolicyErrors.None)
                return true;

            // Разрешаем ошибку цепочки/корня (например, НУЦ Минцифры) строго для публичного iss.moex.com
            return req.RequestUri?.Host.Equals("iss.moex.com", StringComparison.OrdinalIgnoreCase) == true
                   && errors == SslPolicyErrors.RemoteCertificateChainErrors;
        }
    };

    return new HttpClient(handler)
    {
        Timeout = TimeSpan.FromSeconds(15)
    };
}

//

-Первая часть использует проверку https, вторая обходит протокол проверки безопасности. В будущем можно добавить работу системы либо без проверки https (можно оставить ибо только отбирает данные OHLCV), но для формальности лучше предоставлять первый вариант требующий https. В случае проблем российских подписей MOEX и неправильной работы сайта, рекомендуется установить Российские сертификаты "Russian Trusted Sub CA", это решит проблемы, когда офф сайт iss.moex.com в какие то моменты стал небезопасным () из-за чего, терминал ложиться и не может скачать данные о свечах.
